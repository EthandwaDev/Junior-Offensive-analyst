```bash

# View website
python3 -m http.server 8000

# Explore projects
ls -la projects/
tree projects/ # if tree is installed

# View specific project

# Run Python project
cd projects/log-analysis
python3 log_analysis.py

# Compile C++ project
cd projects/cpp-security
g++ -std=c++17 -Wall -Wextra -o secure secure.cpp
./secure
```
